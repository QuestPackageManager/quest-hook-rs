1use std::sync::OnceLock;
2
3use retour::RawDetour;
4
5use crate::{HookName, Priority, UninstallError};
6
7/// A function hook that works across most platforms
8///
9/// Only a single hook may be installed per target; `name` and `priority`
10/// are accepted for API parity with other backends but otherwise ignored
11#[derive(Debug)]
12pub struct FunctionHook {
13    detour: OnceLock<RawDetour>,
14}
15
16impl FunctionHook {
17    /// Creates a new, unitialized hook
18    pub const fn new() -> Self {
19        Self {
20            detour: OnceLock::new(),
21        }
22    }
23
24    /// Installes the hook by redirecting `target` to `hook`, returning true on
25    /// success
26    ///
27    /// # Safety
28    /// `target` and `hook` must have the same signature and calling convention
29    pub unsafe fn install(
30        &self,
31        target: *const (),
32        hook: *const (),
33        _name: HookName,
34        _priority: Priority,
35    ) -> bool {
36        match RawDetour::new(target, hook) {
37            Ok(detour) if detour.enable().is_ok() => {
38                self.detour.set(detour).ok();
39                true
40            }
41            _ => false,
42        }
43    }
44
45    /// Uninstalls the hook
46    ///
47    /// # Safety
48    /// No other thread may be currently executing inside the hook or
49    /// original functions in a way that assumes this hook remains installed
50    pub unsafe fn uninstall(&self) -> Result<(), UninstallError> {
51        match self.detour.get() {
52            Some(detour) if detour.is_enabled() => {
53                detour.disable().map_err(|_err| UninstallError::Failed)
54            }
55            _ => Err(UninstallError::NotInstalled),
56        }
57    }
58
59    /// Whether the hook is installed
60    pub fn is_installed(&self) -> bool {
61        self.detour.get().is_some_and(RawDetour::is_enabled)
62    }
63
64    /// Returns the address of a trampoline function to the original target, if
65    /// installed
66    pub fn original(&self) -> Option<*const ()> {
67        self.detour.get().map(|d| d.trampoline() as *const ())
68    }
69}
70
71impl Default for FunctionHook {
72    fn default() -> Self {
73        Self::new()
74    }
75}
