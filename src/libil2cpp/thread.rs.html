1//! Helpers for running Rust code on native threads that talk to il2cpp.
2//!
3//! Attaching a thread registers it with il2cpp's GC so the GC can scan its
4//! stack for references when deciding what's still reachable. A thread that
5//! isn't attached is invisible to that scan: a GC-managed object referenced
6//! only from a local on such a thread (a [`Gc<T>`](crate::Gc) held in a
7//! local, say) can be collected out from under it while a collection runs
8//! concurrently. Threads Unity itself spawns (the main thread, Unity
9//! job-system workers, ...) are already attached; a thread *you* spawn
10//! (`std::thread::spawn`, a native callback arriving on some other
11//! library's thread, ...) is not, and should attach first if it's going to
12//! hold GC references across a potential collection. Ports beatsaber-hook's
13//! `threading.hpp`.
14//!
15//! Unlike upstream's `attach_thread`, this only attaches to **il2cpp** -
16//! it doesn't also attach the thread to a JVM (`JNIEnv::AttachCurrentThread`
17//! on Android), since this crate has no JNI/JVM handle to do that with.
18//! Calling into JNI from a thread attached here still needs its own
19//! `AttachCurrentThread` call.
20//!
21//! See [`crate::r#async`] for an `.await`-able equivalent of
22//! [`spawn_attached`], built on top of [`attached_invoke`].
23
24use std::thread::{self, JoinHandle};
25
26use crate::raw;

28/// RAII guard produced by [`attach`] - detaches the thread it was created
29/// for when dropped (including when unwinding from a panic).
30pub struct AttachedThread {
31    thread: &'static mut raw::Il2CppThread,
32}
33
34impl Drop for AttachedThread {
35    fn drop(&mut self) {
36        unsafe { raw::thread_detach(self.thread) };
37    }
38}
39
40/// Attaches the current thread to the il2cpp runtime, returning a guard
41/// that detaches it again on drop.
42///
43/// # Safety
44/// The calling thread must not already be attached - attaching an
45/// already-attached thread, or detaching one still in use elsewhere (by
46/// dropping a second, overlapping guard for it), is not a scenario il2cpp
47/// is documented to handle gracefully.
48pub unsafe fn attach() -> AttachedThread {
49    let domain = unsafe { raw::domain_get() };
50    let thread = unsafe { raw::thread_attach(domain) }.expect("il2cpp_thread_attach returned null");
51    AttachedThread { thread }
52}
53
54/// Runs `f` with the current thread attached to il2cpp for the duration,
55/// detaching again once `f` returns (or panics) - see [`attach`].
56///
57/// # Safety
58/// Same as [`attach`]: the calling thread must not already be attached.
59pub unsafe fn attached_invoke<F, T>(f: F) -> T
60where
61    F: FnOnce() -> T,
62{
63    let _guard = unsafe { attach() };
64    f()
65}
66
67/// Spawns a new OS thread that attaches itself to il2cpp before running
68/// `f` and detaches again before exiting - the safe, ergonomic entry point
69/// built on [`attached_invoke`] (safe because a freshly spawned OS thread
70/// is never already attached, satisfying its safety requirement by
71/// construction).
72///
73/// See [`crate::r#async::il2cpp_async`] for a version that returns an
74/// `.await`-able [`Future`](std::future::Future) instead of a
75/// [`JoinHandle`].
76pub fn spawn_attached<F, T>(f: F) -> JoinHandle<T>
77where
78    F: FnOnce() -> T + Send + 'static,
79    T: Send + 'static,
80{
81    thread::spawn(move || unsafe { attached_invoke(f) })
82}
