1//! Bulk introspection helpers for exploring the loaded il2cpp domain - ports
2//! beatsaber-hook's `debug.hpp` (`log_classes`/`log_methods`/`log_fields`/
3//! `log_properties`), useful for reverse-engineering during mod development.
4//! Per-item [`Debug`](std::fmt::Debug)/[`Display`](std::fmt::Display) impls
5//! already exist on [`Il2CppClass`]/[`MethodInfo`]/[`FieldInfo`]/
6//! [`PropertyInfo`]; what's missing without this module is a way to walk
7//! *all* of them at once.
8
9use crate::{raw, Il2CppClass, WrapRaw};
10
11/// Iterates every class loaded in the current domain (across every loaded
12/// assembly's image), optionally restricted to those whose
13/// [`Display`](std::fmt::Display) form (`Namespace::Name`, or
14/// `Declaring/Nested` for a nested type) starts with `prefix`.
15pub fn classes(prefix: Option<&str>) -> impl Iterator<Item = &'static Il2CppClass> + '_ {
16    let domain = unsafe { raw::domain_get() };
17    let mut assemblies_count = 0;
18    let assemblies = unsafe { raw::domain_get_assemblies(domain, &mut assemblies_count) };
19
20    assemblies
21        .iter()
22        .take(assemblies_count)
23        .filter_map(|assembly| unsafe { raw::assembly_get_image(assembly) })
24        .flat_map(|image| {
25            let count = unsafe { raw::image_get_class_count(image) };
26            (0..count).filter_map(move |i| unsafe { raw::image_get_class(image, i) })
27        })
28        .map(|class| unsafe { Il2CppClass::wrap(class) })
29        .filter(move |class| match prefix {
30            Some(prefix) => class.to_string().starts_with(prefix),
31            None => true,
32        })
33}
34
35/// Logs every class in the domain (see [`classes`]), one per line.
36pub fn log_classes(prefix: Option<&str>) {
37    for class in classes(prefix) {
38        debug!("{class}");
39    }
40}
41
42/// Logs every method belonging to `class`.
43pub fn log_methods(class: &Il2CppClass) {
44    for method in class.methods() {
45        debug!("{method}");
46    }
47}
48
49/// Logs every field belonging to `class`.
50pub fn log_fields(class: &Il2CppClass) {
51    for field in class.fields() {
52        debug!("{field:?}");
53    }
54}
55
56/// Logs every property belonging to `class`.
57pub fn log_properties(class: &Il2CppClass) {
58    for property in class.properties() {
59        debug!("{property:?}");
60    }
