1//! Resolution of the Boehm GC functions used internally by libil2cpp.
2//!
3//! [`super::gc_alloc_fixed`]/[`super::gc_free_fixed`] (declared alongside
4//! the rest of libil2cpp's exported functions in `functions.rs`) are
5//! exported symbols, resolved and cached the same way as every other
6//! `il2cpp_` function in this crate - no disassembly needed.
7//! `GarbageCollector::AllocateFixed` isn't exported on unity2018/il2cpp_v24
8//! though, so there it falls back to the deeper, disassembly-based lookup
9//! in [`crate::xref`], available only when the `xref` feature is enabled.
10
11use std::ffi::c_void;
12
13#[cfg(any(feature = "il2cpp_v29", feature = "il2cpp_v31"))]
14use super::gc_alloc_fixed;
15use super::gc_free_fixed;
16
17/// GcAllocFixed allocates a fixed-size object in the Boehm GC heap. The object
18/// is not movable and will not be collected until it is explicitly freed.
19pub type GcAllocFixedFn = unsafe extern "C" fn(size: usize) -> *mut c_void;
20/// GcFreeFixed frees a fixed-size object allocated with GcAllocFixed. The
21/// object must not be used after it is freed.
22pub type GcFreeFixedFn = unsafe extern "C" fn(obj: *mut c_void);
23
24/// `il2cpp_functions!` gives us a safe, `OnceLock`-cached Rust-ABI wrapper,
25/// but [`GcFunctions`] stores plain `extern "C"` pointers so `GcAllocator`
26/// can call xref-resolved and exported-symbol functions the same way. This
27/// is the thin shim that bridges the two for `gc_alloc_fixed`.
28#[cfg(any(feature = "il2cpp_v29", feature = "il2cpp_v31"))]
29unsafe extern "C" fn gc_alloc_fixed_shim(size: usize) -> *mut c_void {
30    unsafe { gc_alloc_fixed(size) }
31}
32
33/// Same as [`gc_alloc_fixed_shim`], for `gc_free_fixed`.
34unsafe extern "C" fn gc_free_fixed_shim(obj: *mut c_void) {
35    unsafe { gc_free_fixed(obj) };
36}
37
38/// Boehm GC functions used by libil2cpp. `gc_alloc_fixed` is `None` if it
39/// couldn't be resolved (only possible on unity2018/il2cpp_v24 without the
40/// `xref` feature).
41pub struct GcFunctions {
42    /// `GarbageCollector::AllocateFixed` (or equivalent) function pointer.
43    pub gc_alloc_fixed: Option<GcAllocFixedFn>,
44    /// `GarbageCollector::FreeFixed` (or equivalent) function pointer.
45    pub gc_free_fixed: Option<GcFreeFixedFn>,
46}
47static GC_FUNCTIONS: std::sync::OnceLock<GcFunctions> = std::sync::OnceLock::new();
48
49impl GcFunctions {
50    #[allow(unused_variables, unused_mut, unused_assignments)]
51    fn resolve(libil2cpp: &[u8]) -> Self {
52        let mut gc_alloc_fixed = None;
53        #[cfg(any(feature = "il2cpp_v29", feature = "il2cpp_v31"))]
54        {
55            gc_alloc_fixed = Some(gc_alloc_fixed_shim as GcAllocFixedFn);
56        }
57        // unity2018/il2cpp_v24 don't export `GarbageCollector::AllocateFixed`
58        // directly - it can only be found by tracing branches (or, failing
59        // that, signature-scanning), which needs the `xref` feature.
60        #[cfg(all(any(feature = "unity2018", feature = "il2cpp_v24"), feature = "xref"))]
61        {
62            gc_alloc_fixed = crate::xref::arch::gc::find_gc_alloc_fixed(libil2cpp);
63        }
64
65        let gc_free_fixed = Some(gc_free_fixed_shim as GcFreeFixedFn);
66
67        Self {
68            gc_alloc_fixed,
69            gc_free_fixed,
70        }
71    }
72
73    /// Resolve and cache the [`GcFunctions`] instance for the given libil2cpp
74    /// binary. This should be called once at the start of the program,
75    /// before any other libil2
76    pub fn init(libil2cpp: &[u8]) -> &'static Self {
77        GC_FUNCTIONS.get_or_init(|| Self::resolve(libil2cpp))
78    }
79
80    /// Get the cached [`GcFunctions`] instance, if it has been initialized.
81    pub fn get() -> Option<&'static Self> {
82        GC_FUNCTIONS.get()
83    }
84}
