19/// An il2cpp type
20#[repr(transparent)]
21pub struct Il2CppType(raw::Il2CppType);

23unsafe impl Send for Il2CppType {}
24unsafe impl Sync for Il2CppType {}

26impl Il2CppType {
27    /// Class of the type
28    pub fn class(&self) -> &Il2CppClass {
29        unsafe { Il2CppClass::wrap(raw::class_from_il2cpp_type(self.raw())) }
30    }
31
32    /// Name of the type
33    pub fn name(&self) -> Cow<'_, str> {
34        if let Some(name) = self.as_builtin().map(Builtin::name) {
35            return name.into();
36        }
37
38        let name = unsafe { raw::type_get_name(self.raw()) };
39        assert!(!name.is_null());
40        unsafe { CStr::from_ptr(name) }.to_string_lossy()
41    }
42
43    /// Whether the type is a ref type
44    pub fn is_ref(&self) -> bool {
45        self.raw().byref() != 0
46    }
47
48    /// [`Il2CppReflectionType`] which represents the type
49    pub fn reflection_object(&self) -> &Il2CppReflectionType {
50        unsafe { Il2CppReflectionType::wrap_mut(raw::type_get_object(self.raw())) }
51    }
52}

85macro_rules! builtins {
86    ($($const:ident => ($variant:ident, $id:ident, $name:literal),)*) => {
87
88        // essentially Windows clang will use i32
89        // https://github.com/rust-lang/rust-bindgen/issues/1966
90
91
92        #[doc = "Builtin C# types"]
93        #[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
94        #[cfg_attr(feature = "il2cpp_v31", repr(u32))]
95        #[cfg_attr(feature = "il2cpp_v29", repr(u32))]
96        #[cfg_attr(feature = "il2cpp_v24", repr(u32))]
97        #[cfg_attr(feature = "unity2018", repr(i32))]
98        pub enum Builtin {
99            $(
100                #[doc = concat!("`", $name, "`")]
101                $variant = $const as u32,
102            )*
103        }
104
105        impl Il2CppType {
106            #[doc = "Whether the type represents the given [`Builtin`]"]
107            #[inline]
108            pub fn is_builtin(&self, builtin: Builtin) -> bool {
109                #[cfg(feature = "il2cpp_v31")]
110                { self.raw().type_() == (builtin as u32).try_into().unwrap() }
111
112                #[cfg(feature = "il2cpp_v29")]
113                { self.raw().type_() == (builtin as u32).try_into().unwrap() }
114
115                #[cfg(feature = "il2cpp_v24")]
116                { self.raw().type_() == builtin as u32 }
117
118                #[cfg(feature = "unity2018")]
119                { self.raw().type_() == builtin as i32 }
120            }
121
122            paste! {
123                $(
124                    #[doc = concat!("Whether the type represents a `", $name , "`")]
125                    pub fn [<is_ $id>](&self) -> bool {
126                        self.is_builtin(Builtin::$variant)
127                    }
128                )*
129            }
130
131            #[doc = "[`Builtin`] the type represents, if any"]
132            pub fn as_builtin(&self) -> Option<Builtin> {
133                #[allow(non_upper_case_globals)]
134                match self.raw().type_() {
135                    $($const => Some(Builtin::$variant),)*
136                    _ => None
137                }
138            }
139        }
140
141        impl Builtin {
142            #[doc = "Name of the builtin"]
143            pub fn name(self) -> &'static str {
144                match self {
145                    $(Self::$variant => $name,)*
146                }
147            }
148        }
149    }
150}

176/// Object used for reflection of types
177#[repr(transparent)]
178pub struct Il2CppReflectionType(#[allow(dead_code)] raw::Il2CppReflectionType);
179
180unsafe impl WrapRaw for Il2CppReflectionType {
181    type Raw = raw::Il2CppReflectionType;
182}
183
184impl Il2CppReflectionType {
185    /// [`Il2CppType`] which this object represents
186    pub fn ty(&self) -> &Il2CppType {
187        unsafe { Il2CppType::wrap(&*self.raw().type_) }
188    }
189
190    /// Instanciates a generic type template with the provided generic
191    /// arguments
192    pub fn make_generic<G>(&self) -> Result<Option<&Self>, Gc<Il2CppException>>
193    where
194        G: Generics,
195    {
196        let generics = G::type_array();
197        let make_generic = self
198            .class()
199            .find_method_unchecked("MakeGenericType", 2)
200            .unwrap();
201        let ret = unsafe {
202            make_generic.invoke_raw(
203                null_mut(),
204                [
205                    self as *const Self as *mut c_void,
206                    (generics as *mut raw::Il2CppArray).cast(),
207                ]
208                .as_mut(),
209            )
210        };
211        let obj = match ret {
212            Ok(Some(obj)) => obj,
213            Ok(None) => return Ok(None),
214            Err(e) => return Err(unsafe { Gc::from(Il2CppException::wrap_mut(e)) }),
215        };
216        let ty = unsafe { &mut *(obj as *mut raw::Il2CppObject).cast() };
217        Ok(Some(ty))
218    }
219}
