26impl Il2CppType {
27    /// Class of the type
28    pub fn class(&self) -> &Il2CppClass {
29        unsafe { Il2CppClass::wrap(raw::class_from_il2cpp_type(self.raw())) }
30    }

32    /// Name of the type
33    pub fn name(&self) -> Cow<'_, str> {
34        if let Some(name) = self.as_builtin().map(Builtin::name) {
35            return name.into();
36        }
37
38        let name = unsafe { raw::type_get_name(self.raw()) };
39        assert!(!name.is_null());
40        unsafe { CStr::from_ptr(name) }.to_string_lossy()
41    }
42
43    /// Whether the type is a ref type
44    pub fn is_ref(&self) -> bool {
45        self.raw().byref() != 0
46    }
47
48    /// [`Il2CppReflectionType`] which represents the type
49    pub fn reflection_object(&self) -> &Il2CppReflectionType {
50        unsafe { Il2CppReflectionType::wrap_mut(raw::type_get_object(self.raw())) }
51    }

53    /// The 0-based position of this type's generic parameter in its owning
54    /// generic class/method's parameter list, if this type actually
55    /// represents an unresolved generic parameter placeholder - a class's
56    /// own `T` in `Foo<T>` (`IL2CPP_TYPE_VAR`), or a method's own `T` in
57    /// `Bar<T>()` (`IL2CPP_TYPE_MVAR`) - rather than a concrete type.
58    /// `None` for any concrete type.
59    ///
60    /// This is what makes matching a generic method's *un-instantiated*
61    /// parameters against caller-supplied types possible at all - see
62    /// [`Il2CppClass::find_method`](crate::Il2CppClass::find_method)'s
63    /// generic-method substitution.
64    pub fn generic_parameter_index(&self) -> Option<u16> {
65        let ty = self.raw().type_();
66        if ty != Il2CppTypeEnum_IL2CPP_TYPE_VAR && ty != Il2CppTypeEnum_IL2CPP_TYPE_MVAR {
67            return None;
68        }
69
70        #[cfg(any(feature = "il2cpp_v31", feature = "il2cpp_v29"))]
71        {
72            // The union holds a handle - a pointer straight into il2cpp's
73            // mmap'd global metadata file, at this parameter's own metadata
74            // entry, whose `num` field already gives the 0-based position
75            // directly (see `raw::Il2CppGenericParameter`).
76            let handle = unsafe { self.raw().data.genericParameterHandle };
77            Some(unsafe { (*handle.cast::<raw::Il2CppGenericParameter>()).num })
78        }
79
80        #[cfg(any(feature = "il2cpp_v24", feature = "unity2018"))]
81        {
82            // No handle indirection on these versions - the union holds the
83            // index directly.
84            Some(unsafe { self.raw().data.genericParameterIndex } as u16)
85        }
86    }
87}

120macro_rules! builtins {
121    ($($const:ident => ($variant:ident, $id:ident, $name:literal),)*) => {
122
123        // essentially Windows clang will use i32
124        // https://github.com/rust-lang/rust-bindgen/issues/1966
125
126
127        #[doc = "Builtin C# types"]
128        #[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
129        #[cfg_attr(feature = "il2cpp_v31", repr(u32))]
130        #[cfg_attr(feature = "il2cpp_v29", repr(u32))]
131        #[cfg_attr(feature = "il2cpp_v24", repr(u32))]
132        #[cfg_attr(feature = "unity2018", repr(i32))]
133        pub enum Builtin {
134            $(
135                #[doc = concat!("`", $name, "`")]
136                $variant = $const as u32,
137            )*
138        }
139
140        impl Il2CppType {
141            #[doc = "Whether the type represents the given [`Builtin`]"]
142            #[inline]
143            pub fn is_builtin(&self, builtin: Builtin) -> bool {
144                #[cfg(feature = "il2cpp_v31")]
145                { self.raw().type_() == (builtin as u32).try_into().unwrap() }
146
147                #[cfg(feature = "il2cpp_v29")]
148                { self.raw().type_() == (builtin as u32).try_into().unwrap() }
149
150                #[cfg(feature = "il2cpp_v24")]
151                { self.raw().type_() == builtin as u32 }
152
153                #[cfg(feature = "unity2018")]
154                { self.raw().type_() == builtin as i32 }
155            }
156
157            paste! {
158                $(
159                    #[doc = concat!("Whether the type represents a `", $name , "`")]
160                    pub fn [<is_ $id>](&self) -> bool {
161                        self.is_builtin(Builtin::$variant)
162                    }
163                )*
164            }
165
166            #[doc = "[`Builtin`] the type represents, if any"]
167            pub fn as_builtin(&self) -> Option<Builtin> {
168                #[allow(non_upper_case_globals)]
169                match self.raw().type_() {
170                    $($const => Some(Builtin::$variant),)*
171                    _ => None
172                }
173            }
174        }
175
176        impl Builtin {
177            #[doc = "Name of the builtin"]
178            pub fn name(self) -> &'static str {
179                match self {
180                    $(Self::$variant => $name,)*
181                }
182            }
183        }
184    }
185}

211/// Object used for reflection of types
212#[repr(transparent)]
213pub struct Il2CppReflectionType(#[allow(dead_code)] raw::Il2CppReflectionType);
214
215unsafe impl WrapRaw for Il2CppReflectionType {
216    type Raw = raw::Il2CppReflectionType;
217}
218
219impl Il2CppReflectionType {
220    /// [`Il2CppType`] which this object represents
221    pub fn ty(&self) -> &Il2CppType {
222        unsafe { Il2CppType::wrap(&*self.raw().type_) }
223    }
224
225    /// Instanciates a generic type template with the provided generic
226    /// arguments
227    pub fn make_generic<G>(&self) -> Result<Option<&Self>, Gc<Il2CppException>>
228    where
229        G: Generics,
230    {
231        self.make_generic_with(&G::classes())
232    }
233
234    /// Instanciates a generic type template with generic arguments found at
235    /// runtime (e.g. via [`Il2CppClass::find`](crate::Il2CppClass::find))
236    /// rather than a compile-time `G: Generics`
237    pub fn make_generic_with(
238        &self,
239        classes: &[&'static Il2CppClass],
240    ) -> Result<Option<&Self>, Gc<Il2CppException>> {
241        let make_generic = self
242            .class()
243            .find_method_unchecked("MakeGenericType", 2)
244            .unwrap();
245        let mut generics = Il2CppArray::<Self>::of_refs(
246            classes.iter().map(|class| class.ty().reflection_object()),
247        );
248        let ret = unsafe {
249            make_generic.invoke_raw(
250                null_mut(),
251                [
252                    self as *const Self as *mut c_void,
253                    generics.get_pointer_mut().cast(),
254                ]
255                .as_mut(),
256            )
257        };
258        let obj = match ret {
259            Ok(Some(obj)) => obj,
260            Ok(None) => return Ok(None),
261            Err(e) => return Err(unsafe { Gc::from(Il2CppException::wrap_mut(e)) }),
262        };
263        let ty = unsafe { &mut *(obj as *mut raw::Il2CppObject).cast() };
264        Ok(Some(ty))
265    }
266}

291mod tests {
292    use std::mem;

294    use super::*;

296    fn leak<T>(value: T) -> &'static T {
297        Box::leak(Box::new(value))
298    }

300    fn fake_generic_param_type(kind: raw::Il2CppTypeEnum, num: u16) -> raw::Il2CppType {
301        let mut param: raw::Il2CppGenericParameter = unsafe { mem::zeroed() };
302        param.num = num;
303        let handle = (leak(param) as *const raw::Il2CppGenericParameter).cast();
304
305        let mut ty: raw::Il2CppType = unsafe { mem::zeroed() };
306        ty.set_type(kind);
307        ty.data.genericParameterHandle = handle;
308        ty
309    }
310
311    #[test]
312    fn generic_parameter_index_reads_an_mvar_placeholders_position() {
313        let raw = fake_generic_param_type(Il2CppTypeEnum_IL2CPP_TYPE_MVAR, 2);
314        let ty = unsafe { Il2CppType::wrap(&raw) };
315        assert_eq!(ty.generic_parameter_index(), Some(2));
316    }
317
318    #[test]
319    fn generic_parameter_index_reads_a_var_placeholders_position() {
320        let raw = fake_generic_param_type(Il2CppTypeEnum_IL2CPP_TYPE_VAR, 0);
321        let ty = unsafe { Il2CppType::wrap(&raw) };
322        assert_eq!(ty.generic_parameter_index(), Some(0));
323    }
324
325    #[test]
326    fn generic_parameter_index_is_none_for_a_concrete_type() {
327        let mut raw: raw::Il2CppType = unsafe { mem::zeroed() };
328        raw.set_type(raw::Il2CppTypeEnum_IL2CPP_TYPE_I4);
329        let ty = unsafe { Il2CppType::wrap(&raw) };
330        assert_eq!(ty.generic_parameter_index(), None);
331    }
