6/// Trait implemented by Rust types that are also C# types
7///
8/// # Safety
9/// The Rust and C# types must be ABI-compatible and the trait implementation
10/// must be correct
11pub unsafe trait Type: 'static {
12    /// Type of the values held in variables of the type
13    type Held<'a>;
14    /// Non-generic version of [`Held`].
15    type HeldRaw;
16
17    /// Namespace containingthe class the type represents
18    const NAMESPACE: &'static str;
19    /// Name of the class the type represents
20    const CLASS_NAME: &'static str;
21
22    /// [`Il2CppClass`] of the type
23    fn class() -> &'static Il2CppClass {
24        Il2CppClass::find(Self::NAMESPACE, Self::CLASS_NAME)
25            .unwrap_or_else(|| panic!("Class {}.{} not found", Self::NAMESPACE, Self::CLASS_NAME))
26    }
27
28    /// Returns the [`Il2CppType`] of this type
29    fn type_() -> &'static Il2CppType {
30        Self::class().ty()
31    }
32
33    /// Whether the type can be used as the implicit `this` argument when
34    /// Rust code calls the given C# instance [`MethodInfo`] - i.e. whether
35    /// `Self` is (or derives from) the method's declaring class.
36    ///
37    /// "Argument" means Rust is the caller here, supplying `this` to native
38    /// code; see [`matches_this_parameter`](Type::matches_this_parameter)
39    /// for the reverse direction.
40    fn matches_this_argument(method: &MethodInfo) -> bool {
41        method.class().is_assignable_from(Self::class())
42    }
43
44    /// Whether the type can receive the implicit `this` argument when C#
45    /// code calls into Rust as the given instance [`MethodInfo`]'s
46    /// implementation.
47    ///
48    /// The callee-side mirror of
49    /// [`matches_this_argument`](Type::matches_this_argument) - the
50    /// assignability check is reversed because caller and callee swap roles
51    /// (here `Self` is the expected/declared type and `method`'s class is
52    /// the value being checked against it, rather than the other way
53    /// around).
54    fn matches_this_parameter(method: &MethodInfo) -> bool {
55        Self::class().is_assignable_from(method.class())
56    }
57
58    /// Whether a *reference* to the type (a GC pointer, e.g. a
59    /// [`Gc<T>`](crate::Gc)) can be passed as an argument of the given
60    /// [`Il2CppType`] when Rust calls a C# method.
61    ///
62    /// "Argument" means Rust is the caller, supplying the value to native
63    /// code. True C# reference types (classes, boxed values, strings, ...) are
64    /// always passed by pointer and so implement this by checking
65    /// assignability alone, ignoring [`Il2CppType::is_ref`] entirely (see
66    /// [`unsafe_impl_reference_type!`](crate::unsafe_impl_reference_type)).
67    ///
68    /// A C# *value* type parameter declared `ref`/`out`/`in` is passed by
69    /// pointer too, though, so [`ValueType`](crate::ValueType)s also
70    /// implement this - but only when `ty.is_ref()` is set, since an
71    /// ordinary (non-byref) value-type parameter must go by value instead
72    /// (see [`matches_value_argument`](Type::matches_value_argument)).
73    fn matches_reference_argument(ty: &Il2CppType) -> bool;
74
75    /// Whether a *value* of the type (the raw struct bytes, not a pointer)
76    /// can be passed as an argument of the given [`Il2CppType`] when Rust
77    /// calls a C# method.
78    ///
79    /// Only meaningful for [`ValueType`](crate::ValueType)s, which match
80    /// when `ty` is *not* declared `ref`/`out`/`in` (see
81    /// [`Il2CppType::is_ref`]) and is assignable - a byref value-type
82    /// parameter is passed by pointer instead and is matched by
83    /// [`matches_reference_argument`](Type::matches_reference_argument).
84    /// True reference types unconditionally return `false` here, since their
85    /// representation is never raw bytes at the ABI boundary.
86    fn matches_value_argument(ty: &Il2CppType) -> bool;
87
88    /// Whether a *reference* to the type can be received as a parameter of
89    /// the given [`Il2CppType`] when C# calls into Rust.
90    ///
91    /// The callee-side mirror of
92    /// [`matches_reference_argument`](Type::matches_reference_argument) -
93    /// see that method for what "reference" means and why
94    /// [`ValueType`](crate::ValueType)s only implement this when
95    /// [`ty.is_ref()`](Il2CppType::is_ref) is set. As with
96    /// [`matches_this_parameter`](Type::matches_this_parameter), the
97    /// assignability check is reversed relative to the argument-side
98    /// version because caller and callee swap roles.
99    fn matches_reference_parameter(ty: &Il2CppType) -> bool;
100
101    /// Whether a *value* of the type can be received as a parameter of the
102    /// given [`Il2CppType`] when C# calls into Rust.
103    ///
104    /// The callee-side mirror of
105    /// [`matches_value_argument`](Type::matches_value_argument) - see that
106    /// method for why only non-byref [`ValueType`](crate::ValueType)s match
107    /// and why true reference types always return `false` here.
108    fn matches_value_parameter(ty: &Il2CppType) -> bool;
109
110    /// Whether the type can hold the value of the given [`Il2CppType`] when
111    /// it comes back as the return value of a C# method Rust called.
112    ///
113    /// Rust is the caller here, receiving the value; see
114    /// [`matches_return`](Type::matches_return) for the callee-side check
115    /// used when a Rust function is invoked as a C# method's implementation.
116    fn matches_returned(ty: &Il2CppType) -> bool {
117        Self::class().is_assignable_from(ty.class())
118    }
119
120    /// Whether the type can be returned as the given [`Il2CppType`] when
121    /// this Rust function is invoked as a C# method's implementation, handing
122    /// the value back to native code.
123    ///
124    /// The callee-side mirror of [`matches_returned`](Type::matches_returned)
125    /// - the assignability check is reversed for the same reason as
126    /// [`matches_this_parameter`](Type::matches_this_parameter).
127    fn matches_return(ty: &Il2CppType) -> bool {
128        ty.class().is_assignable_from(Self::class())
129    }
130}

132// implement type for pointers
133unsafe impl<T: Type> Type for *mut T {
134    type Held<'a> = Option<&'a mut T>;
135    type HeldRaw = *mut T;
136
137    const NAMESPACE: &'static str = T::NAMESPACE;
138    const CLASS_NAME: &'static str = T::CLASS_NAME;
139
140    fn matches_reference_argument(ty: &Il2CppType) -> bool {
141        T::matches_reference_argument(ty)
142    }
143
144    fn matches_value_argument(ty: &Il2CppType) -> bool {
145        T::matches_value_argument(ty)
146    }
147
148    fn matches_reference_parameter(ty: &Il2CppType) -> bool {
149        T::matches_reference_parameter(ty)
150    }
151
152    fn matches_value_parameter(ty: &Il2CppType) -> bool {
153        T::matches_value_argument(ty)
154    }
155}
