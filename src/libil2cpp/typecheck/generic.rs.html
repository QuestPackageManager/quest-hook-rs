1use crate::{Il2CppClass, Type};

3/// Trait implemented for Rust types which can represent a list of C# generic
4/// arguments
5pub trait Generics {
6    /// Number of generic arguments
7    const COUNT: usize;
8
9    /// The generic arguments' classes, in order - used to substitute a
10    /// generic method's un-instantiated parameter types (`T`, `U`, ...) with
11    /// concrete ones for matching, before it's actually been instantiated
12    /// with [`MethodInfo::make_generic`](crate::MethodInfo::make_generic).
13    ///
14    /// Returns `Vec` rather than a `[&'static Il2CppClass; Self::COUNT]`
15    /// array - sizing an array by an associated const like `Self::COUNT` in
16    /// a trait method signature needs the `generic_const_exprs` feature,
17    /// which is unstable enough (frequent ICEs, incomplete) that it's not
18    /// worth enabling crate-wide just for this; the allocation here is
19    /// negligible next to the FFI calls already happening alongside it.
20    /// [`Arguments<const N: usize>`](crate::Arguments) sidesteps the same
21    /// problem by parameterizing the *trait* by `N` instead of using an
22    /// associated const, which would work here too, but would mean
23    /// threading a second const-generic through every `G: Generics` call
24    /// site (`find_generic`, `make_generic`,
25    /// [`Il2CppClass::find_method`](crate::Il2CppClass::find_method), ...).
26    fn classes() -> Vec<&'static Il2CppClass>;
27}
28
29impl Generics for () {
30    const COUNT: usize = 0;
31
32    fn classes() -> Vec<&'static Il2CppClass> {
33        Vec::new()
34    }
35}
36
37impl<T: Type> Generics for T {
38    const COUNT: usize = 1;
39
40    fn classes() -> Vec<&'static Il2CppClass> {
41        vec![T::class()]
42    }
43}
