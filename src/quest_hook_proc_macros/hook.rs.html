12/// The parsed argument list of a `#[hook(...)]` attribute: either the three
13/// required target identifiers, or (mutually exclusively) a single path to a
14/// real method on a class, e.g. `SceneManager::SetActiveScene` - either way
15/// followed by any number of optional `key = "value"` arguments.
16pub enum HookArgs {
17    /// `#[hook("Namespace", "Class", "Method", ...)]`
18    Explicit {
19        namespace: LitStr,
20        class: LitStr,
21        method: LitStr,
22        extra: Vec<HookArg>,
23    },
24    /// `#[hook(SomeClass::method, ...)]`
25    Target { target: Path, extra: Vec<HookArg> },
26}
27
28impl Parse for HookArgs {
29    fn parse(input: ParseStream<'_>) -> Result<Self, Error> {
30        enum Leading {
31            Explicit {
32                namespace: LitStr,
33                class: LitStr,
34                method: LitStr,
35            },
36            Target(Path),
37        }
38
39        // a string literal starts the three-identifier form; anything else
40        // (an identifier or path) must be a path to a method to target
41        // instead
42        let leading = if input.peek(LitStr) {
43            let namespace: LitStr = input.parse()?;
44            input.parse::<Token![,]>()?;
45            let class: LitStr = input.parse()?;
46            input.parse::<Token![,]>()?;
47            let method: LitStr = input.parse()?;
48            Leading::Explicit {
49                namespace,
50                class,
51                method,
52            }
53        } else {
54            Leading::Target(input.parse()?)
55        };
56
57        let mut extra = Vec::new();
58        while input.peek(Token![,]) {
59            input.parse::<Token![,]>()?;
60            if input.is_empty() {
61                break;
62            }
63            extra.push(input.parse()?);
64        }
65
66        Ok(match leading {
67            Leading::Explicit {
68                namespace,
69                class,
70                method,
71            } => Self::Explicit {
72                namespace,
73                class,
74                method,
75                extra,
76            },
77            Leading::Target(target) => Self::Target { target, extra },
78        })
79    }
80}

82pub fn expand(args: &HookArgs, input: ItemFn) -> Result<TokenStream, Error> {
83    let metadata = Metadata::new(args, input)?;
84    metadata.validate()?;
85
86    let outer_fn = metadata.outer_fn();
87    let struct_def = metadata.struct_def();
88    let struct_impl = metadata.struct_impl();
89    let static_def = metadata.static_def();
90    let trait_impl = metadata.trait_impl();
91    let method_check = metadata.method_check();
92
93    let ts = quote! {
94        #outer_fn
95        #struct_def
96        #struct_impl
97        #static_def
98        #trait_impl
99        #method_check
100    };
101    Ok(ts.into())
102}

104/// A single optional `key = "value"` argument in a `#[hook(...)]`
105/// attribute, following the target identifiers.
106pub enum HookArg {
107    /// Overrides the hook's own namespace
108    Namespace(LitStr),
109    /// A `Priority::before` filter
110    Before(LitStr),
111    /// A `Priority::after` filter
112    After(LitStr),
113}
114
115impl Parse for HookArg {
116    fn parse(input: ParseStream<'_>) -> Result<Self, Error> {
117        let key: Ident = input.parse()?;
118        input.parse::<Token![=]>()?;
119        let value: LitStr = input.parse()?;
120
121        match key.to_string().as_str() {
122            "namespace" => Ok(Self::Namespace(value)),
123            "before" => Ok(Self::Before(value)),
124            "after" => Ok(Self::After(value)),
125            other => Err(Error::new_spanned(
126                key,
127                format!("unknown hook argument `{other}`"),
128            )),
129        }
130    }
131}

133/// A parsed `before`/`after` filter value, mirroring
134/// `hook_backend::HookFilter` but with both fields resolved up front instead
135/// of carried around as raw strings.
136#[cfg_attr(test, derive(Debug))]
137enum HookFilter {
138    /// Matches a hook with the given name in any namespace
139    Name(String),
140    /// Matches any hook in the given namespace
141    Namespace(String),
142    /// Matches a hook with the given name in the given namespace
143    Both { namespace: String, name: String },
144}
145
146impl HookFilter {
147    /// Parses a `before`/`after` filter value in one of the forms `"name"`,
148    /// `"namespace::"`, or `"namespace::name"`.
149    fn parse(lit: &LitStr) -> Result<Self, Error> {
150        let value = lit.value();
151        let Some((namespace, name)) = value.split_once("::") else {
152            return Ok(Self::Name(value));
153        };
154        let namespace = (!namespace.is_empty()).then(|| namespace.to_string());
155        let name = (!name.is_empty()).then(|| name.to_string());
156
157        match (namespace, name) {
158            // `"namespace::name"`
159            (Some(namespace), Some(name)) => Ok(Self::Both { namespace, name }),
160            // `"namespace::"`
161            (Some(namespace), None) => Ok(Self::Namespace(namespace)),
162            // `"::"`: neither half names anything to match
163            (None, _) => Err(Error::new_spanned(
164                lit,
165                "filter must specify a namespace, a name, or both",
166            )),
167        }
168    }
169
170    /// Returns a `TokenStream2` that constructs a `hook_backend::HookFilter`
171    fn expr(&self) -> TokenStream2 {
172        let (namespace, name) = match self {
173            Self::Name(name) => (
174                quote!(::std::option::Option::None),
175                quote!(::std::option::Option::Some(#name)),
176            ),
177            Self::Namespace(namespace) => (
178                quote!(::std::option::Option::Some(#namespace)),
179                quote!(::std::option::Option::None),
180            ),
181            Self::Both { namespace, name } => (
182                quote!(::std::option::Option::Some(#namespace)),
183                quote!(::std::option::Option::Some(#name)),
184            ),
185        };
186        quote! {
187            ::quest_hook::hook_backend::HookFilter {
188                namespace: #namespace,
189                name: #name,
190            }
191        }
192    }
193}

195/// Where this hook's target method is: either its namespace/class/method
196/// given directly as string literals, or a path to the real method on a
197/// class implementing `libil2cpp::Type`, e.g. `SceneManager::SetActiveScene`.
198/// `class_path` (`SceneManager`) supplies the namespace/class name and
199/// `method_name` (`"SetActiveScene"`) the method name, while this hook's
200/// declared types get checked against `method_path`'s actual signature (see
201/// [`Metadata::method_check`]).
202enum Location {
203    Explicit {
204        namespace: String,
205        class: String,
206        method: String,
207    },
208    Method {
209        class_path: Path,
210        method_path: Path,
211        method_name: String,
212    },
213}
214
215/// Splits `path` (e.g. `SceneManager::SetActiveScene`) into its class
216/// prefix (`SceneManager`) and final method segment (`SetActiveScene`).
217fn split_method_path(path: &Path) -> Result<(Path, Ident), Error> {
218    let mut class_path = path.clone();
219    let Some(last) = class_path.segments.pop() else {
220        return Err(Error::new_spanned(path, "expected a path to a method"));
221    };
222    // `pop` leaves the `::` that used to separate `last` from the rest
223    // dangling as trailing punctuation
224    class_path.segments.pop_punct();
225
226    if class_path.segments.is_empty() {
227        return Err(Error::new_spanned(
228            path,
229            "expected a path to a method on a class, e.g. `SceneManager::SetActiveScene`",
230        ));
231    }
232
233    Ok((class_path, last.into_value().ident))
234}

239    hook_namespace: Option<String>,

242    /// `Priority::after` filters

247impl Metadata {
248    fn new(args: &HookArgs, input: ItemFn) -> Result<Self, Error> {
249        let (location, extra) = match args {
250            HookArgs::Explicit {
251                namespace,
252                class,
253                method,
254                extra,
255            } => (
256                Location::Explicit {
257                    namespace: namespace.value(),
258                    class: class.value(),
259                    method: method.value(),
260                },
261                extra,
262            ),
263            HookArgs::Target { target, extra } => {
264                let (class_path, method) = split_method_path(target)?;
265                (
266                    Location::Method {
267                        class_path,
268                        method_path: target.clone(),
269                        method_name: method.to_string(),
270                    },
271                    extra,
272                )
273            }
274        };
275
276        let mut hook_namespace = None;
277        let mut before = Vec::new();
278        let mut after = Vec::new();
279
280        for arg in extra {
281            match arg {
282                HookArg::Namespace(value) => {
283                    if hook_namespace.is_some() {
284                        return Err(Error::new_spanned(value, "duplicate hook argument"));
285                    }
286                    hook_namespace = Some(value.value());
287                }
288                HookArg::Before(value) => before.push(HookFilter::parse(value)?),
289                HookArg::After(value) => after.push(HookFilter::parse(value)?),
290            }
291        }
292
293        Ok(Self {
294            location,
295            hook_namespace,
296            before,
297            after,
298            input,
299        })
300    }

363    fn validate(&self) -> Result<(), Error> {
364        if let Some(constness) = self.input.sig.constness {
365            return Err(Error::new_spanned(constness, "Cannot hook const functions"));
366        }
367
368        if let Some(asyncness) = self.input.sig.asyncness {
369            return Err(Error::new_spanned(asyncness, "Cannot hook async functions"));
370        }
371
372        if let Some(Abi {
373            name: Some(abi), ..
374        }) = &self.input.sig.abi
375        {
376            if abi.value() != "C" {
377                return Err(Error::new_spanned(
378                    abi,
379                    "Cannot hook functions with non-C ABIs",
380                ));
381            }
382        }
383
384        let generics = &self.input.sig.generics;
385        if !self
386            .input
387            .sig
388            .generics
389            .params
390            .iter()
391            .all(|g| matches!(g, GenericParam::Lifetime(_)))
392        {
393            return Err(Error::new_spanned(
394                generics,
395                "Cannot hook generic functions",
396            ));
397        }
398
399        for (i, arg) in self.input.sig.inputs.iter().enumerate() {
400            match arg {
401                FnArg::Receiver(_) => {
402                    return Err(Error::new_spanned(
403                        arg,
404                        "Cannot hook functions taking a `self` parameter",
405                    ))
406                }
407                FnArg::Typed(PatType {
408                    pat: box Pat::Ident(ident),
409                    ..
410                }) if ident.ident == "self" => {
411                    return Err(Error::new_spanned(
412                        arg,
413                        "Cannot hook functions taking a `self` parameter",
414                    ))
415                }
416                FnArg::Typed(PatType { attrs, .. }) if i != 0 => {
417                    let has_this_attr = attrs.iter().any(|a| attr_is(a, "this"));
418                    if has_this_attr {
419                        return Err(Error::new_spanned(
420                            arg,
421                            "`this` can only be the first parameter",
422                        ));
423                    }
424                }
425                FnArg::Typed(_) => (),
426            }
427        }
428
429        if let Some(variadic) = &self.input.sig.variadic {
430            return Err(Error::new_spanned(
431                variadic,
432                "Cannot hook variadic functions",
433            ));
434        }
435
436        Ok(())
437    }

459    fn this(&self) -> Option<&PatType> {
460        let first_input = match self.input.sig.inputs.iter().next()? {
461            FnArg::Typed(arg) => arg,
462            FnArg::Receiver(_) => unreachable!(),
463        };
464
465        let is_this = match &first_input.pat {
466            box Pat::Ident(ident) if ident.ident == "this" => true,
467            _ => first_input.attrs.iter().any(|a| attr_is(a, "this")),
468        };
469        if !is_this {
470            return None;
471        }
472
473        Some(first_input)
474    }

558        let unsafety = self.input.sig.unsafety;
559        let name = self.fn_name();
560        let return_ty = self.actual_return_ty();
561        let inner_fn = self.inner_fn();

564            .this_ident()
565            .zip(self.actual_this_ty())

569            .this_ident()

579            .map(|i| quote!(::quest_hook::libil2cpp::Parameter::from_actual(#i),));

585                let r = inner(#this_arg #(#params_args)*);
586                ::quest_hook::libil2cpp::Return::into_actual(r)

615    fn install_fn(&self) -> TokenStream2 {
616        let vis = &self.input.vis;
617
618        let namespace = self.namespace_expr();
619        let class = self.class_expr();
620        let method = self.method_expr();
621
622        let this_ty = self.typechecking_this_ty();
623        let params_ty = self.typechecking_params_ty();
624        let return_ty = self.return_ty();
625
626        let fn_name = self.fn_name();
627        let name = self.hook_name();
628        let hook_namespace = self.hook_namespace_expr();
629        let hook_name_str = self.hook_name().to_string();
630
631        quote! {
632            #vis fn install(&self) -> Result<::quest_hook::HookHandle, quest_hook::HookInstallError> {
633                use ::std::ptr::null_mut;
634                use ::std::sync::atomic::Ordering;
635                use ::quest_hook::HookInstallError;
636                use ::quest_hook::hook_backend::HookName;
637                use ::quest_hook::libil2cpp::{Il2CppClass, WrapRaw};
638
639                if self.hook.is_installed() {
640                    return Err(HookInstallError::AlreadyInstalled);
641                }
642
643                let class = match Il2CppClass::find(#namespace, #class) {
644                    Some(class) => class,
645                    None => return Err(HookInstallError::ClassNotFound),
646                };
647                let method = match class.find_method_callee::<#this_ty, #params_ty, #return_ty>(#method) {
648                    Ok(method) => method,
649                    Err(_) => return Err(HookInstallError::MethodNotFound),
650                };
651
652                let hook_name = HookName { namespace: #hook_namespace, name: #hook_name_str };
653
654                let success = unsafe {
655                    self.hook.install(
656                        method.raw().methodPointer.unwrap() as *const (),
657                        #fn_name as *const (),
658                        hook_name,
659                        self.priority(),
660                    )
661                };
662                if success {
663                    Ok(::quest_hook::HookHandle::new(&#name.hook))
664                } else {
665                    Err(HookInstallError::InstallError)
666                }
667            }
668        }
669    }

692    fn original_fn(&self) -> TokenStream2 {
693        let vis = &self.input.vis;
694        let return_ty = self.return_ty();
695        let original_ty = self.original_ty();
696
697        let this_param = self
698            .this_ident()
699            .zip(self.this_ty())
700            .map(|(i, t)| quote!(#i: #t,));
701
702        let this_arg = self
703            .this_ident()
704            .map(|i| quote!(::quest_hook::libil2cpp::ThisParameter::into_actual(#i),));
705
706        let params_params = self
707            .params_ident()
708            .zip(self.params_ty())
709            .map(|(i, t)| quote!(#i: #t,));
710
711        let params_args = self
712            .params_ident()
713            .map(|i| quote!(::quest_hook::libil2cpp::Parameter::into_actual(#i),));
714
715        quote! {
716            #vis fn original(&self, #this_param #(#params_params)*) -> #return_ty {
717                use ::std::mem::transmute;
718                use ::std::sync::atomic::Ordering;
719
720                let ptr = self.hook.original().expect("hook is not installed");
721                let original = unsafe { transmute::<*const (), #original_ty>(ptr) };
722
723                let r = original(#this_arg #(#params_args)*);
724                ::quest_hook::libil2cpp::Return::from_actual(r)
725            }
726        }
727    }
728
729    fn struct_impl(&self) -> TokenStream2 {
730        let struct_name = self.struct_name();
731        let install_fn = self.install_fn();
732        let priority_fn = self.priority_fn();
733        let original_fn = self.original_fn();
734
735        quote! {
736            impl #struct_name {
737                #install_fn
738                #priority_fn
739                #original_fn
740            }
741        }
742    }
743
744    fn trait_impl(&self) -> TokenStream2 {
745        let struct_name = self.struct_name();
746
747        let namespace = self.namespace_expr();
748        let class = self.class_expr();
749        let method = self.method_expr();
750
751        let this_ty = staticify(self.typechecking_this_ty());
752        let params_ty = staticify(self.typechecking_params_ty());
753        let return_ty = staticify(self.return_ty());
754
755        let fn_name = self.fn_name();
756        let hook_name_str = self.hook_name().to_string();
757        let hook_namespace = self.hook_namespace_expr();
758
759        quote! {
760            impl ::quest_hook::Hook for #struct_name {
761                type This = #this_ty;
762                type Parameters = #params_ty;
763                type Return = #return_ty;
764
765                const NAMESPACE: &'static str = #namespace;
766                const CLASS_NAME: &'static str = #class;
767                const METHOD_NAME: &'static str = #method;
768                const HOOK_NAMESPACE: &'static str = #hook_namespace;
769                const HOOK_NAME: &'static str = #hook_name_str;
770
771                fn install(&self) -> Result<::quest_hook::HookHandle, ::quest_hook::HookInstallError> {
772                    self.install()
773                }
774
775                fn priority(&self) -> ::quest_hook::hook_backend::Priority {
776                    self.priority()
777                }
778
779                fn original(&self) -> Option<*const ()> {
780                    self.hook.original()
781                }
782                fn hook(&self) -> *const () {
783                    #fn_name as *const ()
784                }
785            }
786        }
787    }
788
789    /// Add a compile-time check that this hook's declared types match the target
790    /// method's actual signature, if the target was given as a path to a real
791    /// method instead of as string literals.
792    fn method_check(&self) -> Option<TokenStream2> {
793        let method_path = self.method_path()?;
794
795        let this_ty = self.this_ty().map(|t| quote!(#t,));
796        let params_ty = self.params_ty().map(|t| quote!(#t,));
797        let return_ty = self.return_ty();
798
799        Some(quote! {
800            const _: fn(#this_ty #(#params_ty)*) -> #return_ty = #method_path;
801        })
802    }
803}

807        paren_token: Default::default(),
808        elems: Default::default(),
809    })

816fn staticify(tokens: impl ToTokens) -> TokenStream2 {
817    let mut ts = TokenStream2::new();
818    let mut iter = tokens.to_token_stream().into_iter().peekable();
819    while let Some(tt) = iter.next() {
820        match &tt {
821            TokenTree2::Group(g) => {
822                let delimiter = g.delimiter();
823                let stream = staticify(g.stream());
824                ts.extend_one(TokenTree2::Group(Group::new(delimiter, stream)));
825            }
826            TokenTree2::Punct(p) if p.as_char() == '&' => match iter.peek() {
827                Some(TokenTree2::Punct(p)) if p.as_char() == '\'' => ts.extend_one(tt),
828                _ => ts.extend_one(quote_spanned!(tt.span()=> &'static)),
829            },
830            _ => ts.extend_one(tt),
831        }
832    }
833    ts
834}
