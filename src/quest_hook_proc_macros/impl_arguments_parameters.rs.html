1use std::ops::Range;

3use proc_macro::TokenStream;
4use quote::{format_ident, quote};
5use syn::{Index, Result};

7pub fn expand(range: Range<usize>) -> Result<TokenStream> {
8    let mut ts = TokenStream::new();
9    for n in range {
10        let generic_params_argument = (1..=n).map(|n| format_ident!("A{}", n));
11        let matches_argument = generic_params_argument
12            .clone()
13            .enumerate()
14            .map(|(n, gp)| quote!(<#gp>::matches(params.get_unchecked(#n).ty())));
15        let invokables = (0..n).map(Index::from).map(|n| quote!(self.#n.invokable()));
16
17        let generic_params_parameter = (1..=n).map(|n| format_ident!("P{}", n));
18        let matches_parameter = generic_params_parameter
19            .clone()
20            .enumerate()
21            .map(|(n, gp)| quote!(<#gp>::matches(params.get_unchecked(#n).ty())));
22
23        // log params
24        let log_parameters = generic_params_parameter.clone().enumerate().map(|(n, gp)| {
25            quote!(unsafe {
26                #[cfg(feature = "trace")]
27                crate::debug!("\tChecking parameter {} {:?} vs method param {:?}",
28                    #n,
29                    stringify!(#gp),
30                    params.get(#n).map(|p| (p.ty(), <#gp>::matches(p.ty())))
31                );
32            })
33        });
34
35        let generic_params_argument_tuple = generic_params_argument.clone();
36        let generic_params_argument_where = generic_params_argument.clone();
37        let generic_params_argument_type = generic_params_argument.clone();
38        let generic_params_argument_classes = generic_params_argument.clone();
39
40        let generic_params_parameter_tuple = generic_params_parameter.clone();
41        let generic_params_parameter_where = generic_params_parameter.clone();
42
43        let impl_ts = quote! {
44            unsafe impl<#(#generic_params_argument),*> Arguments<#n> for (#(#generic_params_argument_tuple,)*)
45            where
46                #(#generic_params_argument_where: Argument),*
47            {
48                type Type = (#(#generic_params_argument_type::Type,)*);
49
50                fn matches(method: &MethodInfo) -> bool {
51                    let params = method.parameters();
52                    params.len() == #n && unsafe { #(#matches_argument) && * }
53                }
54
55                fn classes() -> [&'static Il2CppClass; #n] {
56                    [#(#generic_params_argument_classes::class()),*]
57                }
58
59                fn invokable(&mut self) -> [*mut c_void; #n] {
60                    [#(#invokables),*]
61                }
62            }
63
64            unsafe impl<#(#generic_params_parameter),*> Parameters for (#(#generic_params_parameter_tuple,)*)
65            where
66                #(#generic_params_parameter_where: Parameter),*
67            {
68                const COUNT: usize = #n;
69
70                fn matches(method: &MethodInfo) -> bool {
71                    let params = method.parameters();
72                    #(#log_parameters)*;
73                    params.len() == #n && unsafe { #(#matches_parameter) && * }
74                }
75            }
76        };
77        ts.extend(TokenStream::from(impl_ts));
78    }
79
80    Ok(ts)
81}
